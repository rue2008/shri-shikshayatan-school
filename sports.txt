function frame() {
  ctx.clearRect(0,0,W,H);
  dots.forEach(p => {
    const dx = mx-p.x, dy = my-p.y, d = Math.hypot(dx,dy);
    if (d < 180) { p.vx += dx/d*0.03; p.vy += dy/d*0.03; }
    p.vx += (Math.random()-0.5)*0.03; p.vy += (Math.random()-0.5)*0.03 - 0.004;
    p.vx *= 0.97; p.vy *= 0.97;
    p.x = (p.x+p.vx+W)%W; p.y = (p.y+p.vy+H)%H;
    ctx.beginPath();
    ctx.fillStyle = "hsl(" + p.hue + ",90%,70%)";
    ctx.shadowColor = "hsl(" + p.hue + ",100%,60%)";
    ctx.shadowBlur = 14;
    ctx.arc(p.x,p.y,p.r,0,Math.PI*2);
    ctx.fill();
  });
  requestAnimationFrame(frame);
}

function fallbackReply(q) {
  const text = q.toLowerCase();
  const tokens = text.split(/[^a-z0-9]+/).filter(Boolean);
  if (/^(hi|hello|hey|yo)\b/.test(text.trim())) return topics.find(t=>t.words.includes("hi")).replies[0];
  if (/\b(thanks|thank|thx)\b/.test(text)) return "Anytime! Try asking about the Asian Games final or the Lord's Test next.";
  const idx = topics.findIndex(t => t.words.some(w => tokens.some(tok => tok === w || (w.length > 4 && tok.startsWith(w)))));
  if (idx >= 0) {
    const t = topics[idx];
    const n = used[idx] = used[idx] === undefined ? 0 : used[idx] + 1;
    return t.replies[n % t.replies.length];
  }
  return "I don't have a specific answer for that one yet — try asking about a player, strike rate, economy, the Asian Games final, or the Lord's Test.";
}
function typeOut(el, text) {
  if (still) { el.textContent = text; return Promise.resolve(); }
  return new Promise(done => {
    let i = 0;
    const timer = setInterval(() => {
      i += 2;
      el.textContent = text.slice(0, i);
      log.scrollTop = log.scrollHeight;
      if (i >= text.length) { clearInterval(timer); el.textContent = text; done(); }
    }, 16);
  });
}
async function ask(question) {
  question = question.trim();
  if (!question) return;
  addMsg(question, "me");
  input.value = "";
  sendBtn.disabled = true;
  const bubble = addMsg("Thinking...", "bot");
  await new Promise(r => setTimeout(r, 500));
  const answer = fallbackReply(question);
  await typeOut(bubble, answer);
  sendBtn.disabled = false;
  log.scrollTop = log.scrollHeight;
  input.focus();
}
